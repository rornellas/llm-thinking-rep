WITH monthly_sales AS (
    SELECT
        customer_id,
        DATE_TRUNC('month', created_at) AS month,
        SUM(net_amount) AS revenue
    FROM orders
    WHERE status = 'COMPLETED'
    GROUP BY customer_id, DATE_TRUNC('month', created_at)
), ranked AS (
    SELECT
        customer_id,
        month,
        revenue,
        ROW_NUMBER() OVER (PARTITION BY month ORDER BY revenue DESC) AS position
    FROM monthly_sales
)
SELECT customer_id, month, revenue
FROM ranked
WHERE position <= 10;

def stable_softmax(values):
    import numpy as np
    x = np.asarray(values, dtype=np.float64)
    shifted = x - np.max(x, axis=-1, keepdims=True)
    numerator = np.exp(shifted)
    denominator = np.sum(numerator, axis=-1, keepdims=True)
    return numerator / denominator

Subtracting the maximum does not change the normalized probabilities because the common exponential factor cancels. It prevents overflow for large logits, but it does not solve underflow in every extreme case, nor does it address incorrect masking or invalid values.
